11. Wait for "css=[role=dialog] [data-slot=form-message]:text-is('Invalid phone number')"
12. Audit inside "[role=dialog]" as "Customer Details - invalid phone number"
# Active starts ticked; the other three start clear.
13. Click "css=[role=dialog] button[role=checkbox] >> nth=0"
14. Click "css=[role=dialog] button[role=checkbox] >> nth=1"
15. Click "css=[role=dialog] button[role=checkbox] >> nth=2"
16. Click "css=[role=dialog] button[role=checkbox] >> nth=3"

# Valid, with Save sent and held: the form while it is saving.
18. Type "ZZ Not Saved" into "css=[role=dialog] input[name='customerName']"
19. Type "4165551234" into "css=[role=dialog] input[name='phone']"
20. Fail "POST poRequired"
21. Hold "POST poRequired"
22. Click "css=[role=dialog] button:text-is('Save')"
23. Wait for "css=[role=dialog] button[role=checkbox]:disabled"
24. Audit inside "[role=dialog]" as "Customer Details - saving"
25. Release "POST poRequired"
26. Wait for "css=[role=dialog] button[role=checkbox]:enabled"
